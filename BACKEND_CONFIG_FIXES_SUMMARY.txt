                       PYDANTIC V2 UPDATES
================================================================================

Before (Deprecated):
  from pydantic import validator

  @validator('field_name')
  def validate_field(cls, v):
      # validation logic

  class Config:
      schema_extra = {...}

After (Modern):
  from pydantic import field_validator, ConfigDict

  model_config = ConfigDict(
      json_schema_extra={...}
  )

  @field_validator('field_name')
  @classmethod
  def validate_field(cls, v):
      # validation logic
