## Component Props Reference

```typescript
interface PrivacyScorecardProps {
  score: number;                    // 0-100
  level: 'low' | 'medium' | 'high' | 'critical';
  flags: Array<{
    code: string;                   // 'FLAG_TRACKING', etc.
    category: string;               // Category name
    weight: number;                 // 5-25 points
    description: string;            // Human-readable text
  }>;
}

// Usage:
<PrivacyScorecard
  score={74}
  level="medium"
  flags={[
    { code: 'FLAG_TRACKING', category: 'Tracking', weight: 10, 
      description: 'Cross-site tracking detected' },
    { code: 'FLAG_DATA_SELLING', category: 'Data', weight: 15,
      description: 'Data shared with 3rd parties' }
  ]}
/>
```
